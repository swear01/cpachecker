\begin{frame}{Design: \texttt{thinking} Disabled}
  \begin{columns}[c]
    \column{0.42\textwidth}
      {\color{accentorange}
        More rounds within 300\,s\\[0.3em]
        $\Rightarrow$ better PAR-2 than one deep call}\\[0.8em]
      \small Predicates are short C expressions\\
      — no chain-of-thought needed.\\[0.4em]
      Fast call also cuts cost per task.

    \column{0.54\textwidth}
      \renewcommand{\arraystretch}{1.35}
      \begin{tabular}{lrr}
        \toprule
        Mode & Latency & PAR-2 avg \\
        \midrule
        Thinking ON  & 52--88\,s & 225--258\,s \\
        \textbf{Thinking OFF} & \alert{\textbf{2.4\,s}} & \textbf{228\,s} \\
        \bottomrule
      \end{tabular}
  \end{columns}
  \note{DeepSeek V4 Pro defaults to thinking enabled (chain-of-thought), which adds 52–88\,s latency per call and reasoning token cost. VGuide disables it — structured JSON predicate proposals do not need deep CoT; within 300\,s, the 2.4\,s fast-call path allows more scheduling rounds and yields better PAR-2 than a single deep thinking call.}
\end{frame}
